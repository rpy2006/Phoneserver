// Add inside the `android { ... }` block, alongside your existing
// defaultConfig / compileSdk / etc:

android {
    // ... your existing config stays as-is ...

    defaultConfig {
        // ... existing applicationId, minSdk, targetSdk, versionCode/Name ...

        // Limit to the ABIs that actually matter for a phone - cuts build
        // time and APK size significantly compared to building all 4.
        ndk {
            abiFilters "arm64-v8a", "armeabi-v7a"
        }

        externalNativeBuild {
            cmake {
                cppFlags "-std=c++17"
                arguments "-DANDROID_STL=c++_shared"
            }
        }
    }

    externalNativeBuild {
        cmake {
            path "src/main/cpp/CMakeLists.txt"
            version "3.22.1"   // match whatever CMake version AndroidIDE has bundled
        }
    }
}

// dependencies {} block is unchanged - no new Gradle dependency needed for
// this, since llama.cpp is compiled from source via CMake, not pulled as
// a Maven artifact.
